43class BlueairAwsHumidifier(BlueairEntity[BlueairUpdateCoordinatorDeviceAws], HumidifierEntity): 

44 """Controls a standalone humidifier (master power toggles the device).""" 

45 

46 @classmethod 

47 def is_implemented(cls, coordinator: BlueairUpdateCoordinator) -> bool: 

48 """Return whether the coordinator supports this entity.""" 

49 # Standalone humidifiers only. 2-in-1 combo devices (which expose a 

50 # dedicated ``humidifier_mode``) are handled by BlueairAwsComboHumidifier so 

51 # that turning humidification off does not power down the purifier. 

52 return ( 

53 isinstance(coordinator, BlueairUpdateCoordinatorDeviceAws) 

54 and coordinator.auto_regulated_humidity is not NotImplemented 

55 and coordinator.humidifier_mode is NotImplemented 

56 )

108 async def async_set_mode(self, mode: str) -> None: 

109 """Set the mode.""" 

110 if mode == MODE_AUTO: 

111 # This mode doesn't apply when off 

112 await self.coordinator.set_fan_auto_mode(True) 

113 await self.coordinator.set_running(True) 

114 elif mode == MODE_SLEEP: 

115 # This mode doesn't apply when off 

116 await self.coordinator.set_night_mode(True) 

117 await self.coordinator.set_fan_auto_mode(False) 

118 await self.coordinator.set_running(False) 

119 elif mode == MODE_FAN_SPEED: 

120 # This mode doesn't apply when off 

121 await self.coordinator.set_fan_auto_mode(False) 

122 await self.coordinator.set_night_mode(False) 

123 await self.coordinator.set_running(True) 

124 else: 

125 raise ServiceValidationError( 

126 translation_domain=DOMAIN, 

127 translation_key="invalid_mode", 

128 translation_placeholders={"mode": mode}, 

129 )

142class BlueairAwsComboHumidifier(BlueairEntity[BlueairUpdateCoordinatorDeviceAws], HumidifierEntity): 

143 """Controls humidification on a 2-in-1 Purify+Humidify device (e.g. DH3i). 

144 

145 On these devices humidification is a sub-function of the purifier, toggled 

146 via ``humidifier_mode`` independently of the device's master power (``standby``). 

147 Powering the purifier on/off and selecting its operating mode is handled by 

148 the fan entity, so this entity intentionally exposes only humidification 

149 on/off plus the target humidity (no modes). 

150 """ 

151 

152 @classmethod 

153 def is_implemented(cls, coordinator: BlueairUpdateCoordinator) -> bool: 

154 """Return whether the coordinator supports this entity.""" 

155 return ( 

156 isinstance(coordinator, BlueairUpdateCoordinatorDeviceAws) 

157 and coordinator.humidifier_mode is not NotImplemented 

158 ) 

159 

160 def __init__(self, coordinator: BlueairUpdateCoordinatorDeviceAws) -> None: 

161 """Initialize the combo humidifier.""" 

162 self._attr_device_class = HumidifierDeviceClass.HUMIDIFIER 

163 self._attr_supported_features = HumidifierEntityFeature(0) 

164 super().__init__("Humidifier", coordinator, "humidifier") 

165
